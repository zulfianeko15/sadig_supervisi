    function handleLogin() {
      const u = dom.loginUsername.value.trim();
      const p = dom.loginPassword.value.trim();
      if (!u || !p) return;
      
      showLoader();
      google.script.run
        .withSuccessHandler(res => {
          hideLoader();
          if (res.success) {
            currentUser = res.user;
            dom.loginAlert.classList.add('d-none');
            dom.pageLogin.classList.add('d-none');
            dom.pageMain.classList.remove('d-none');
            
            dom.userInfo.innerText = currentUser.nama || currentUser.username;
            
            if (currentUser.role === 'supervisor') {
              if (dom.adminGuruControls) dom.adminGuruControls.style.display = 'none';
              dom.navSettingsItem.classList.add('d-none');
              dom.navUsersItem.classList.add('d-none');
            } else {
              if (dom.adminGuruControls) dom.adminGuruControls.style.display = 'inline';
              dom.navSettingsItem.classList.remove('d-none');
              dom.navUsersItem.classList.remove('d-none');
            }
            
            loadInitialData();
          } else {
            dom.loginAlert.innerText = res.message;
            dom.loginAlert.classList.remove('d-none');
          }
        })
        .withFailureHandler(err => {
          hideLoader();
          Swal.fire('Error', err.message || 'Gagal login!', 'error');
        })
        .checkLogin(u, p);
    }

    function processCSVUpload() {
      const fileInput = document.getElementById('csvFileInput');
      if (!fileInput.files.length) {
        Swal.fire('Peringatan', 'Silakan pilih file CSV terlebih dahulu!', 'warning');
        return;
      }
      const file = fileInput.files[0];
      const reader = new FileReader();
      reader.onload = function(e) {
        const lines = e.target.result.split('\n');
        const guruArray = [];
        for (let i = 1; i < lines.length; i++) {
          const line = lines[i].trim();
          if (line) {
            const cols = line.split(',');
            if (cols.length >= 3) {
              guruArray.push({
                nama: cols[0].trim(),
                nip: cols[1] ? cols[1].trim() : '',
                mapel: cols[2].trim(),
                tglSupervisi: cols[3] ? cols[3].trim() : ''
              });
            }
          }
        }
        if (guruArray.length > 0) {
          showLoader();
          google.script.run
            .withSuccessHandler(res => {
              hideLoader();
              bootstrap.Modal.getInstance(document.getElementById('modalUpload')).hide();
              Swal.fire('Sukses', res.message, 'success');

    function renderTableUsers() {
      const tbody = dom.tblUsersPenggunaBody;
      tbody.innerHTML = '';
      
      if (listUsersData.length === 0) {
        tbody.innerHTML = '<tr><td colspan="6" class="text-center text-muted py-3">Belum ada akun pengguna/supervisor.</td></tr>';
        return;
      }
      
      listUsersData.forEach(u => {
        const tr = document.createElement('tr');
        
        let btnActionHtml = '';
        if (currentUser && currentUser.role === 'admin') {
          let btnDelete = '';
          if (u.username !== 'admin') {
            btnDelete = `<button class="btn btn-sm btn-danger" onclick="deleteUserAkun('${u.username}')"><i class="fas fa-trash me-1"></i>Hapus</button>`;
          }
          btnActionHtml = `
            <div class="d-inline-flex text-nowrap">
              <button class="btn btn-sm btn-warning me-1" onclick="editUserAkun('${u.username}')"><i class="fas fa-edit me-1"></i>Edit</button>
              ${btnDelete}
            </div>
          `;
        } else {
          btnActionHtml = '<span class="badge bg-secondary">Akses Terbatas</span>';
        }
        
        const roleBadge = u.role === 'admin' ? '<span class="badge bg-danger">Administrator</span>' : '<span class="badge bg-primary">Supervisor</span>';
        
        tr.innerHTML = `
          <td class="fw-bold">${u.username}</td>
          <td><code>${u.password}</code></td>
          <td class="fw-semibold">${u.nama}</td>
          <td>${u.nip || '-'}</td>
          <td class="text-center">${roleBadge}</td>
          <td class="text-center">${btnActionHtml}</td>
        `;
        tbody.appendChild(tr);
      });
    }

        'Pengawas Pembina': document.getElementById('setPengawas').value.trim(),

      };
      
      showLoader();
      google.script.run
        .withSuccessHandler(res => {
          hideLoader();
          Swal.fire('Sukses', res.message, 'success');
          loadInitialData();
        })
        .withFailureHandler(err => {
          hideLoader();
          Swal.fire('Error', err.message, 'error');
        })
        .simpanSettings(data);
    }

    function downloadPDFPersiapan() {
      const idGuru = document.getElementById('selGuruPersiapan').value;
      if (!idGuru) {
        Swal.fire('Peringatan', 'Silakan pilih guru terlebih dahulu!', 'warning');
        return;
      }
      const g = listGuruData.find(x => x.id === idGuru);
      const namaGuru = g ? g.nama : 'Guru';
      const fileName = `${namaGuru}_nilai persiapan.pdf`;
      const element = document.getElementById('areaPersiapan');
      const opt = {
        margin: 0.3,
        filename: fileName,
        image: { type: 'jpeg', quality: 0.98 },
        html2canvas: { scale: 2, useCORS: true },
        jsPDF: { unit: 'in', format: 'a4', orientation: 'portrait' }
      };
      html2pdf().set(opt).from(element).save();
    }
